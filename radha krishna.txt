# Drawing functions
def move(x, y):
    pen.penup()
    pen.goto(x, y)
    pen.pendown()

def circle(x, y, radius, color):
    move(x, y - radius)
    pen.color(color)
    pen.begin_fill()
    pen.circle(radius)
    pen.end_fill()

def line(x1, y1, x2, y2, color, width=3):
    move(x1, y1)
    pen.color(color)
    pen.width(width)
    pen.goto(x2, y2)
    pen.width(1)

def ellipse(x, y, rx, ry, color):
    move(x + rx, y)
    pen.color(color)
    pen.begin_fill()
    for i in range(361):
        angle = math.radians(i)
        pen.goto(x + rx * math.cos(angle),
                 y + ry * math.sin(angle))
    pen.end_fill()

def text(x, y, message, size, color):
    move(x, y)
    pen.color(color)
    pen.write(message, align="center",
              font=("Arial", size, "bold"))

def flower(x, y, color):
    for i in range(8):
        angle = math.radians(i * 45)
        px = x + 12 * math.cos(angle)
        py = y + 12 * math.sin(angle)
        circle(px, py, 6, color)
    circle(x, y, 5, "#ffdf75")

import turtle

screen = turtle.Screen()
screen.setup(800, 700)
screen.bgcolor("#fff4df")
screen.title("Radha and Krishna")

pen = turtle.Turtle()
pen.speed(0)
pen.hideturtle()
pen.pensize(3)


def circle(x, y, radius, color, outline="#4b3028"):
    pen.penup()
    pen.goto(x, y - radius)
    pen.pendown()
    pen.color(outline, color)
    pen.begin_fill()
    pen.circle(radius)
    pen.end_fill()


def polygon(points, color, outline="#4b3028"):
    pen.penup()
    pen.goto(points[0])
    pen.pendown()
    pen.color(outline, color)
    pen.begin_fill()
    for point in points[1:]:
        pen.goto(point)
    pen.goto(points[0])
    pen.end_fill()


def line(points, color="#4b3028", width=3):
    pen.penup()
    pen.goto(points[0])
    pen.pendown()
    pen.pencolor(color)
    pen.pensize(width)
    for point in points[1:]:
        pen.goto(point)
    pen.pensize(3)


def draw_body(x, dress_color, trim_color):
    # Flowing dress
    polygon([
        (x - 55, -10), (x + 55, -10), (x + 105, -220),
        (x - 105, -220)
    ], dress_color)

    # Decorative hem
    line([(x - 100, -190), (x, -205), (x + 100, -190)],
         trim_color, 5)

    # Neck and necklace
    polygon([(x - 20, 15), (x + 20, 15), (x + 15, -28),
             (x - 15, -28)], "#f2c6a5", "#4b3028")
    line([(x - 25, -5), (x, -25), (x + 25, -5)],
         "#ffd44f", 4)
    circle(x, -26, 5, "#e29c26")


def draw_face(x, skin_color, hair_color):
    # Hair behind the face
    circle(x, 75, 58, hair_color)

    # Face
    circle(x, 70, 43, skin_color)

    # Eyes
    circle(x - 15, 79, 4, "#241811", "#241811")
    circle(x + 15, 79, 4, "#241811", "#241811")

    # Eyebrows, nose, and smile
    line([(x - 23, 91), (x - 14, 95)], "#4b3028", 2)
    line([(x + 14, 95), (x + 23, 91)], "#4b3028", 2)
    line([(x, 76), (x - 3, 62), (x + 2, 60)], "#9b5b43", 2)
    line([(x - 10, 48), (x, 44), (x + 10, 48)], "#8b3d3d", 2)


def draw_crown(x, color):
    polygon([
        (x - 42, 112), (x - 32, 157), (x - 10, 135),
        (x, 170), (x + 12, 135), (x + 34, 157), (x + 43, 112)
    ], color)
    line([(x - 38, 116), (x + 38, 116)], "#ffd44f", 4)
    circle(x, 146, 6, "#fff0a8")


# Bodies first, so the heads appear in front
